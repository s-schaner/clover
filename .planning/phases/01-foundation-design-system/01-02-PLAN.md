**Replace app/layout.tsx** with the complete root layout that loads all three fonts:

```tsx
import type { Metadata } from 'next';
import { Space_Grotesk, Inter, JetBrains_Mono } from 'next/font/google';
import './globals.css';

const spaceGrotesk = Space_Grotesk({
  subsets: ['latin'],
  variable: '--font-space-grotesk',
  display: 'swap',
});

const inter = Inter({
  subsets: ['latin'],
  variable: '--font-inter',
  display: 'swap',
});

const jetbrainsMono = JetBrains_Mono({
  subsets: ['latin'],
  variable: '--font-jetbrains-mono',
  display: 'swap',
});

export const metadata: Metadata = {
  title: 'Clover Labs',
  description: 'We build the systems other firms call impossible.',
};

export default function RootLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  return (
    <html
      lang="en"
      className={`${spaceGrotesk.variable} ${inter.variable} ${jetbrainsMono.variable}`}
      data-scroll-behavior="smooth"
    >
      <body className="font-body">{children}</body>
    </html>
  );
}
```

Key requirements:
- All three fonts use `variable` option to inject CSS custom properties
- All use `display: 'swap'` to prevent layout shift
- `data-scroll-behavior="smooth"` on html — this is REQUIRED in Next.js 16 (replaces the removed automatic smooth scroll override)
- `className="font-body"` on body to apply Inter as default body font via the Tailwind utility
- metadata.title = 'Clover Labs', metadata.description = tagline
